
# Task 9 — find


# Navigate to Task 9
cd ~/module-1-linux-basics/09-find

# Check current directory
pwd

# List directory contents
ls

# List files recursively
find .

# Find all regular files
find . -type f



# 1. Find all .log files


find . -type f -name "*.log"

# Find .log files inside a specific directory
find logs -type f -name "*.log"



# 2. Find recently modified files


# Files modified within the last day
find . -type f -mtime -1

# Files modified within the last 60 minutes
find . -type f -mmin -60



# 3. Find files larger than a particular size


# Files larger than 50 KB
find . -type f -size +50k

# Files larger than 1 MB
find . -type f -size +1M

# Files larger than 50 MB
find . -type f -size +50M



# 4. Find files older than a particular number
#    of days


# Create a test file
touch test-delete/old-file.txt

# Change its modification time to 5 days ago
touch -d "5 days ago" test-delete/old-file.txt

# Verify modification time
ls -l test-delete/old-file.txt

# Find files older than 3 days
find . -type f -mtime +3



# 5. Find files with specific permissions


# Create a test file
touch config/secret.conf

# Set permissions to 600
chmod 600 config/secret.conf

# Verify permissions
ls -l config/secret.conf

# Find files with exactly 600 permissions
find . -type f -perm 600



# 6. Find files with a particular name

find . -type f -name "report.txt"

# Find all configuration files
find . -type f -name "*.conf"

# Find all text files
find . -type f -name "*.txt"



# 7. Find files owned by a particular user


# Display current username
whoami

# Find files owned by afrooz
find . -type f -user afrooz



# 8. find with -exec


# Display permissions, filename and owner
# for every .log file
find . -type f -name "*.log" -exec stat -c "%A %n %U" {} \;

# Explanation:
# {}  = current file found by find
# \;  = end of the -exec command

# 9. Safe deletion demonstration


# List disposable test files
ls test-delete

# First inspect files that match the condition
find test-delete -type f -name "*.log" -print

# Delete matching .log files only inside test-delete
find test-delete -type f -name "*.log" -delete

# Verify deletion
ls test-delete



# 10. Important safety practice


# ALWAYS inspect results first
find test-delete -type f -name "*.log" -print

# Only after verifying the output should a
# destructive operation be considered.

# Common mistakes corrected during the task


# WRONG:
find . -type -f

# CORRECT:
find . -type f


# WRONG:
find . -type f "*.log"

# CORRECT:
find . -type f -name "*.log"


# WRONG:
find . -type -mtime +3

# CORRECT:
find . -type f -mtime +3


# WRONG:
find . -type f user afrooz

# CORRECT:
find . -type f -user afrooz


# WRONG:
find . -type f -name "*.log" -exec stat -c "%A %n %U" {} ;/

# CORRECT:
find . -type f -name "*.log" -exec stat -c "%A %n %U" {} \;
